1. Unión cruzada
SELECT e.last_name, d.department_name
FROM hr.employees e
CROSS JOIN hr.departments d;

2. Combinación natural (departments + locations)
SELECT department_id, department_name, location_id, city
FROM hr.departments
NATURAL JOIN hr.locations;

3. Natural join con filtro (departments + locations, solo 20 y 50)
SELECT department_id, department_name, location_id, city
FROM hr.departments
NATURAL JOIN hr.locations
WHERE department_id IN (20, 50);

4. Join usando location_id (solo ubicación 1400)
SELECT d.department_id, d.department_name, d.location_id, l.city
FROM hr.departments d
JOIN hr.locations l
ON d.location_id = l.location_id
WHERE d.location_id = 1400;

5. Departamentos 10, 20 y 30 en Seattle
SELECT l.city, d.department_name, d.location_id, d.department_id
FROM hr.departments d
JOIN hr.locations l
ON d.location_id = l.location_id
WHERE l.city = 'Seattle'
AND d.department_id IN (10, 20, 30);

6. Empleados con cargos que ganan más de 12,000
SELECT e.first_name, e.last_name, e.hire_date, e.job_id, j.job_title, j.max_salary
FROM hr.employees e
JOIN hr.jobs j
ON e.job_id = j.job_id
WHERE j.max_salary > 12000;

7. Empleados (con o sin departamento asignado)
SELECT e.first_name, e.last_name, d.department_name
FROM hr.employees e
LEFT JOIN hr.departments d
ON e.department_id = d.department_id;

8. Departamentos (con o sin empleados asignados)
SELECT e.first_name, e.last_name, d.department_name
FROM hr.employees e
RIGHT JOIN hr.departments d
ON e.department_id = d.department_id;

9. Empleados y departamentos, incluyendo los no asignados en ambos
SELECT e.first_name, e.last_name, d.department_name
FROM hr.employees e
FULL OUTER JOIN hr.departments d
ON e.department_id = d.department_id;

10. Empleado y su jefe
SELECT e.last_name AS Employee,
       e.employee_id AS Emp#,
       m.last_name AS Manager,
       m.employee_id AS Mgr#
FROM hr.employees e
JOIN hr.employees m
ON e.manager_id = m.employee_id;

11. Todos los empleados (aunque no tengan jefe), ordenados por apellido
SELECT e.last_name AS Employee,
       e.employee_id AS Emp#,
       m.last_name AS Manager,
       m.employee_id AS Mgr#
FROM hr.employees e
LEFT JOIN hr.employees m
ON e.manager_id = m.employee_id
ORDER BY e.last_name;

12. Empleados contratados antes que sus jefes
SELECT e.last_name AS Employee,
       e.hire_date AS "Emp Hired",
       m.last_name AS Manager,
       m.hire_date AS "Mgr Hired"
FROM hr.employees e
JOIN hr.employees m
ON e.manager_id = m.employee_id
WHERE e.hire_date < m.hire_date;

13. Jerarquía del departamento de Lex De Haan
SELECT e.last_name, e.salary, e.department_id
FROM hr.employees e
WHERE e.department_id = (
    SELECT department_id
    FROM hr.employees
    WHERE last_name = 'De Haan'
);

14. Diagrama jerárquico completo de empleados
SELECT LPAD(' ', LEVEL*2) || last_name AS "Employee",
       employee_id,
       manager_id,
       job_id
FROM hr.employees
START WITH manager_id IS NULL
CONNECT BY PRIOR employee_id = manager_id;
